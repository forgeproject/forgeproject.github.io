Game & Players

Get Player
local player = game:GetPlayer("Name")
if player then
  print(player.GamerID)
end
Get All Players
local players = {}
game:GetAllPlayers(players)
for _, p in ipairs(players) do
  print(p.GamerID)
end
Send Notification
game:AddNotification(
  "Message",
  Color.Green,
  NotifyRecipient.AllPlayers
)
Block Mined Event
game:AddEventBlockMined(
  Block.None,
  function(block, aux, pos, hand)
    print("Mined " .. block)
  end
)
Block Placed Event
game:AddEventBlockPlaced(
  Block.None,
  function(block, pos, hand)
    print("Placed " .. block)
  end
)
Check Multiplayer
if game.IsMultiplayer then
  print("Multiplayer game")
end

if game.IsHost then
  print("I'm the host")
end

Actor - Health & Status

Damage Actor
actor:TakeDamageAndDisplay(
  DamageType.Physical,
  10,
  actor.Position
)
Strike Attack
local strike = StrikeInfo.new()
strike.Damage = 25
strike.Type = DamageType.Physical
local dmg = 0
target:Strike(attacker, strike, dmg)
Check Health
print("Health: " .. actor.Health)
print("Max Health: " .. actor.MaxHealth)

if actor.Health < 20 then
  print("Low health!")
end
Check Status
if actor.IsDeadOrInactiveOrDisabled then
  return  -- Skip inactive actors

if actor.IsOnGround then
  print("On ground")
end

if actor.IsCrouching then
  print("Crouching")
end

actor:TeleportTo(
  Vector3.new(0, 50, 0)

Map & Blocks

Get Block ID
local map = game.Map
local point = map:GetPoint(
  Vector3.new(0, 0, 0)
)
local block = map:GetBlockID(point)
print("Block: " .. block)
Convert Position
-- Vector to Point
local point = map:GetPoint(
  Vector3.new(100, 50, 100)
)

-- Point to Vector
local vec = map:GetPosition(point)
Check Valid Point
if map:IsValidPoint(point) then
  print("In bounds")
else
  print("Out of bounds")
end
Common Blocks
  • Block.Stone
  • Block.Dirt
  • Block.Grass
  • Block.Wood
  • Block.Obsidian
Commit Changes
-- After modifying blocks
map:Commit()  -- Update visuals
Block Cache Tips
-- Use normal for many nearby blocks
map:GetBlockID(point)

-- Use NoCache for scattered blocks
map:GetBlockIDNoCache(point)

Hooks - Modifiers

Double Damage
local Hook = {}
function Hook:OnModifyDamageDealt(
  actor, target, weapon, damage
)
  damage.Damage = damage.Damage * 2
end
actor.Hooks:AddHook(Hook)
Health Boost
local Hook = {}
function Hook:OnModifyMaxHealth(
  actor, maxHealth
)
  maxHealth.Value = maxHealth.Value + 100
end
actor.Hooks:AddHook(Hook)
Triple Jump
local Hook = {}
function Hook:OnModifyMaxJumpCount(
  actor, jumpCount
)
  jumpCount.Value = 3
end
actor.Hooks:AddHook(Hook)
Double Speed
local Hook = {}
function Hook:OnModifyMovementSpeed(
  actor, speed
)
  speed.Value = speed.Value * 2
end
actor.Hooks:AddHook(Hook)
Double XP
local Hook = {}
function Hook:OnModifyXPGain(
  actor, skill, item, target, xp
)
  xp.Value = xp.Value * 2
end
actor.Hooks:AddHook(Hook)
Add Hook with Priority
-- Priority 0 runs first
actor.Hooks:AddHook(MyHook, 0)

-- Remove hook
actor.Hooks:RemoveHook(MyHook)

Environment & Weather

Add Rain
local envMgr = game.EnvManager
local point = GlobalPoint3D.new(0, 0, 0)
envMgr:AddRain(
  point,
  0.8,      -- density
  0.5,      -- size
  1.0, 0.5, -- wind
  Color.Blue,
  true      -- network
)
Add Fog
envMgr:AddFog(
  point,
  0.5,        -- density
  10, 100,    -- start/end dist
  1.0,        -- vertical density
  Color.Gray,
  5000,       -- duration (ms)
  true        -- network
)
Add Hail
envMgr:AddHail(
  point,
  0.8,        -- density
  0.5,        -- size
  1.0, 0.5,   -- wind
  Color.White,
  0.5, 1.0,   -- hail size/gravity
  true
)
Remove Weather
envMgr:RemoveRain(point, true)
envMgr:RemoveFog(point, true)
envMgr:RemoveHail(point, true)

Common Patterns

Safe Player Loop
local players = {}
game:GetAllPlayers(players)
for _, player in ipairs(players) do
  if not player.IsDeadOrInactiveOrDisabled then
    -- Safe to use player
  end
end
Check & Get Player
local player = game:GetPlayer("Name")
if player and
   not player.IsDeadOrInactiveOrDisabled
then
  -- Safe to interact
end
Safe Health Check
if actor.Health > 0 and
   not actor.IsDeadOrInactiveOrDisabled
then
  print("Actor alive: " .. actor.Health)
end
Vector Distance
local dist = Vector3.Distance(
  actor.Position,
  target.Position
)
if dist < 50 then
  print("Close to target")
end
